// ========== READ SOIL SENSOR ==========
int readSoilSensor() {
  long total = 0;
  for (int i = 0; i < 10; i++) {
    total += analogRead(SOIL_PIN);
    delay(10);
  }
  return total / 10;
}

// ========== CALCULATE VALUES FROM RAW READINGS ==========
void computeDerived() {
  if (soilDry != soilWet) {
    soilPercent = map(soilRaw, soilDry, soilWet, 0, 100);
  }
  soilPercent = constrain(soilPercent, 0, 100);
  rainDetected = (rainRaw < rainThreshold);
}

// ========== READ ALL SENSORS ==========
void readSensors() {

  float t = dht.readTemperature();
  float h = dht.readHumidity();

  if (isnan(t) || isnan(h)) {
    dhtError = true;
  } else {
    temperature = t;
    humidity = h;
    dhtError = false;
  }

  soilRaw = readSoilSensor();
  rainRaw = analogRead(RAIN_PIN);

  computeDerived();

  Record r;
  r.temp = temperature;
  r.hum = humidity;
  r.soil = soilPercent;
  r.rain = rainDetected ? 1 : 0;

  if (historyCount < MAX_HISTORY) {
    history[historyCount] = r;
    historyCount++;
  } else {
    for (int i = 1; i < MAX_HISTORY; i++) history[i - 1] = history[i];
    history[MAX_HISTORY - 1] = r;
  }

  Serial.println();
  Serial.println("--------------------------------------------");

  if (dhtError) {
    Serial.println("Temperature: ERROR (check DHT11)");
    Serial.println("Humidity   : ERROR (check DHT11)");
  } else {
    Serial.print("Temperature: "); Serial.print(temperature, 1); Serial.println(" C");
    Serial.print("Humidity   : "); Serial.print(humidity, 1);    Serial.println(" %");
  }

  Serial.print("Soil       : "); Serial.print(soilPercent); Serial.println(" %");
  Serial.print("Rain       : "); Serial.println(rainDetected ? "DETECTED" : "NOT DETECTED");
  Serial.print("Website    : http://"); Serial.println(WiFi.localIP());
  Serial.println("--------------------------------------------");
}

// ========== URL ENCODE (for Telegram text) ==========
String urlEncode(const String &s) {
  String out = "";
  char buf[4];
  for (size_t i = 0; i < s.length(); i++) {
    char c = s[i];
    if (isalnum((unsigned char)c) || c == '-' || c == '_' || c == '.' || c == '~') {
      out += c;
    } else {
      snprintf(buf, sizeof(buf), "%%%02X", (unsigned char)c);
      out += buf;
    }
  }
  return out;
}

// ========== SEND TELEGRAM MESSAGE ==========
bool sendTelegram(const String &message) {
  if (WiFi.status() != WL_CONNECTED) return false;

  WiFiClientSecure client;
  client.setInsecure();      // skips certificate check (simple, less secure)

  HTTPClient http;
  String url = "https://api.telegram.org/bot" + String(tgToken) + "/sendMessage";

  http.setTimeout(10000);
  if (!http.begin(client, url)) return false;

  http.addHeader("Content-Type", "application/x-www-form-urlencoded");
  String body = "chat_id=" + String(tgChatId) + "&text=" + urlEncode(message);

  int code = http.POST(body);
  Serial.print("Telegram HTTP code: ");
  Serial.println(code);
  if (code != 200) Serial.println(http.getString());   // shows the error reason

  http.end();
  return (code == 200);
}

// ========== TELEGRAM ALERT CHECK ==========
void checkTelegramAlert(bool alert) {
  unsigned long now = millis();

  // send only when an alert STARTS, and respect the cooldown
  if (alert && !lastAlertState && (lastAlertSent == 0 || now - lastAlertSent >= ALERT_COOLDOWN)) {

    String msg = "ALERT - Smart Agriculture\n";
    if (rainDetected) msg += "- Rainfall detected\n";
    if (soilPercent < soilAlertLevel) msg += "- Soil moisture low: " + String(soilPercent) + "%\n";
    if (!dhtError) {
      msg += "Temp: " + String(temperature, 1) + " C, Humidity: " + String(humidity, 0) + "%\n";
    }
    msg += "Dashboard: http://" + WiFi.localIP().toString();

    if (sendTelegram(msg)) lastAlertSent = now;
  }

  lastAlertState = alert;
}

// ========== BUZZER AND LED ==========
void updateAlerts() {

  bool alert = rainDetected || soilPercent < soilAlertLevel;

  checkTelegramAlert(alert);      // Telegram is NOT affected by LED/Buzzer modes

  // ----- LED: AUTO follows alert, ON always on, OFF always off -----
  ledState = (ledMode == 1) || (ledMode == 0 && alert);
  digitalWrite(LED_PIN, ledState ? HIGH : LOW);

  // ----- BUZZER: AUTO follows alert, ON always beeps, OFF silent -----
  bool buzzerActive = (buzMode == 1) || (buzMode == 0 && alert);

  unsigned long now = millis();

  if (!buzzerActive) {
    digitalWrite(BUZZER_PIN, LOW);
    buzzerOn = false;
    return;
  }

  if (buzzerOn && now - beepStarted >= BEEP_DURATION) {
    digitalWrite(BUZZER_PIN, LOW);
    buzzerOn = false;
  }

  if (!buzzerOn && now - lastBeep >= BEEP_INTERVAL) {
    digitalWrite(BUZZER_PIN, HIGH);
    buzzerOn = true;
    beepStarted = now;
    lastBeep = now;
  }
}

void sendData() {

  json.reserve(5500);

  json = "{";

  json += "\"temperature\":" + String(temperature, 1) + ",";
  json += "\"humidity\":" + String(humidity, 1) + ",";
  json += "\"soil\":" + String(soilPercent) + ",";
  json += "\"soilRaw\":" + String(soilRaw) + ",";
  json += "\"rainRaw\":" + String(rainRaw) + ",";

  json += "\"rain\":" + String(rainDetected ? 1 : 0) + ",";
  json += "\"dhtError\":" + String(dhtError ? 1 : 0) + ",";

  bool alert = rainDetected || soilPercent < soilAlertLevel;

  json += "\"alert\":" + String(alert ? 1 : 0) + ",";
  json += "\"led\":" + String(ledState ? 1 : 0) + ",";
  json += "\"buzzer\":" + String(buzzerOn ? 1 : 0) + ",";

  unsigned long elapsed = millis() - lastSample;
  unsigned long nextIn = (elapsed >= SAMPLE_INTERVAL) ? 0 : (SAMPLE_INTERVAL - elapsed) / 1000;

  json += "\"next\":" + String(nextIn) + ",";
  json += "\"interval\":" + String(SAMPLE_INTERVAL / 1000) + ",";
  json += "\"refresh\":" + String(WEBSITE_INTERVAL / 1000) + ",";
  json += "\"uptime\":" + String(millis() / 1000) + ",";
  json += "\"rssi\":" + String(WiFi.RSSI()) + ",";

  json += "\"cfg\":{";
  json += "\"tempLow\":" + String(tempLow) + ",";
  json += "\"tempHigh\":" + String(tempHigh) + ",";
  json += "\"humLow\":" + String(humLow) + ",";
  json += "\"humHigh\":" + String(humHigh) + ",";
  json += "\"soilAlert\":" + String(soilAlertLevel) + ",";
  json += "\"rainThr\":" + String(rainThreshold) + ",";
  json += "\"soilDry\":" + String(soilDry) + ",";
  json += "\"soilWet\":" + String(soilWet) + ",";
  json += "\"ledMode\":" + String(ledMode) + ",";
  json += "\"buzMode\":" + String(buzMode);
  json += "},";

  json += "\"history\":[";

  for (int i = 0; i < historyCount; i++) {
    if (i > 0) json += ",";
    json += "{";
    json += "\"temp\":" + String(history[i].temp, 1) + ",";
    json += "\"hum\":" + String(history[i].hum, 1) + ",";
    json += "\"soil\":" + String(history[i].soil) + ",";
    json += "\"rain\":" + String(history[i].rain);
    json += "}";
  }

  json += "]}";

  server.sendHeader("Cache-Control", "no-store");
  server.send(200, "application/json", json);
}

// ========== CHANGE THRESHOLDS LIVE ==========
void handleSet() {

  if (server.hasArg("tempLow"))   tempLow        = constrain(server.arg("tempLow").toInt(), -10, 60);
  if (server.hasArg("tempHigh"))  tempHigh       = constrain(server.arg("tempHigh").toInt(), -10, 80);
  if (server.hasArg("humLow"))    humLow         = constrain(server.arg("humLow").toInt(), 0, 100);
  if (server.hasArg("humHigh"))   humHigh        = constrain(server.arg("humHigh").toInt(), 0, 100);
  if (server.hasArg("soilAlert")) soilAlertLevel = constrain(server.arg("soilAlert").toInt(), 0, 100);
  if (server.hasArg("rainThr"))   rainThreshold  = constrain(server.arg("rainThr").toInt(), 0, 4095);

  saveSettings();
  computeDerived();

  server.send(200, "application/json", "{\"ok\":1}");
}

// ========== LED / BUZZER MODE FROM THE WEBSITE ==========
void handleMode() {

  if (server.hasArg("led")) ledMode = constrain(server.arg("led").toInt(), 0, 2);
  if (server.hasArg("buz")) buzMode = constrain(server.arg("buz").toInt(), 0, 2);

  saveSettings();
  updateAlerts();     // apply immediately

  server.send(200, "application/json", "{\"ok\":1}");
}

// ========== SOIL CALIBRATION FROM THE WEBSITE ==========
void handleCal() {

  String type = server.arg("type");

  int raw = readSoilSensor();
  soilRaw = raw;

  bool ok = false;

  if (type == "dry" && raw > soilWet + 100) {
    soilDry = raw;
    ok = true;
  } else if (type == "wet" && raw < soilDry - 100) {
    soilWet = raw;
    ok = true;
  }

  if (ok) {
    saveSettings();
    computeDerived();
    server.send(200, "application/json", "{\"ok\":1,\"raw\":" + String(raw) + "}");
  } else {
    server.send(400, "application/json", "{\"ok\":0,\"raw\":" + String(raw) + "}");
  }
}

function setMeter(p, value, min, max, lo, hi, state) {
  const pct = v => Math.max(0, Math.min(100, (v - min)/(max - min)*100));
  const s = $(p + "Safe"), m = $(p + "Mark");
  s.style.left = pct(lo) + "%";
  s.style.width = Math.max(0, pct(hi) - pct(lo)) + "%";

}

const charts = {};

function agoText(samples) {
  const s = samples * IV;
  if (s === 0) return "now";
  if (s >= 60) return "-" + (Math.round(s/6)/10) + "m";
  return "-" + s + "s";
}

function drawChart(id) {
  const s = charts[id]; if (!s) return;
  const cv = $(id), ctx = cv.getContext("2d");
  const o = s.o, v = s.v, n = v.length;

  const w = cv.clientWidth, h = 220, dpr = window.devicePixelRatio || 1;
  cv.width = w * dpr; cv.height = h * dpr;
  ctx.setTransform(dpr,0,0,dpr,0,0); ctx.clearRect(0,0,w,h);

  const L = 34, R = 10, T = 12, B = 26;
  const cw = w - L - R, ch = h - T - B;
  const X = i => L + (n < 2 ? cw/2 : i*cw/(n-1));
  const Y = val => T + ch - (Math.max(o.min, Math.min(o.max, val)) - o.min)/(o.max - o.min)*ch;
  s.geo = { L:L, cw:cw };

  // safe-range band
  if (o.band) {
    const y1 = Y(o.band[1]), y2 = Y(o.band[0]);
    ctx.fillStyle = COL.band;
    ctx.fillRect(L, y1, cw, Math.max(0, y2 - y1));
  }

  // grid + y labels
  ctx.font = "11px " + FONT; ctx.lineWidth = 1;
  for (let i = 0; i <= 4; i++) {
    const y = T + ch*i/4;
    ctx.strokeStyle = COL.line;
    ctx.beginPath(); ctx.moveTo(L,y); ctx.lineTo(w-R,y); ctx.stroke();
    ctx.fillStyle = COL.muted; ctx.textAlign = "right";
    ctx.fillText(Math.round(o.min + (o.max - o.min)*(4-i)/4), L - 6, y + 4);
  }

  // x labels
  ctx.textAlign = "center"; ctx.fillStyle = COL.muted;
  if (n >= 2) {
    [0,0.25,0.5,0.75,1].forEach(f => {
      ctx.fillText(agoText(Math.round((n-1)*(1-f))), L + cw*f, h - 8);
    });
  } else ctx.fillText("now", L + cw/2, h - 8);

  // limit lines
  (o.lines || []).forEach(t => {
    if (t.v === undefined || t.v === null) return;
    const y = Y(t.v);
    ctx.setLineDash([5,5]); ctx.strokeStyle = t.c; ctx.lineWidth = 1.2;
    ctx.beginPath(); ctx.moveTo(L,y); ctx.lineTo(w-R,y); ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle = t.c; ctx.textAlign = "right";
    ctx.fillText(t.l, w - R - 4, y - 4);
  });

  if (n === 0) {
    ctx.fillStyle = COL.muted; ctx.textAlign = "center"; ctx.font = "13px " + FONT;
    ctx.fillText("Waiting for data...", w/2, h/2);
    return;
  }

  // smooth path
  function path() {
    ctx.beginPath();
    ctx.moveTo(X(0), Y(v[0]));
    for (let i = 1; i < n; i++) {
      const xm = (X(i-1) + X(i))/2, ym = (Y(v[i-1]) + Y(v[i]))/2;
      ctx.quadraticCurveTo(X(i-1), Y(v[i-1]), xm, ym);
    }
    ctx.lineTo(X(n-1), Y(v[n-1]));
  }

  // area
  const grad = ctx.createLinearGradient(0, T, 0, T + ch);
  grad.addColorStop(0, o.color + "40"); grad.addColorStop(1, o.color + "05");
  path(); ctx.lineTo(X(n-1), T+ch); ctx.lineTo(X(0), T+ch); ctx.closePath();
  ctx.fillStyle = grad; ctx.fill();

  // line
  path();
  ctx.strokeStyle = o.color; ctx.lineWidth = 2.2; ctx.lineJoin = "round"; ctx.lineCap = "round";
  ctx.stroke();

  // min / max markers
  if (n > 2) {
    let iMin = 0, iMax = 0;
    v.forEach((x,i) => { if (x < v[iMin]) iMin = i; if (x > v[iMax]) iMax = i; });
    [iMin, iMax].forEach(k => {
      ctx.fillStyle = COL.bg; ctx.strokeStyle = o.color; ctx.lineWidth = 2;
      ctx.beginPath(); ctx.arc(X(k), Y(v[k]), 3.5, 0, Math.PI*2); ctx.fill(); ctx.stroke();
    });
  }

  // out-of-range dots
  if (o.band) {
    ctx.fillStyle = COL.red;
    v.forEach((x,i) => {
      if (x < o.band[0] || x > o.band[1]) {
        ctx.beginPath(); ctx.arc(X(i), Y(x), 2.6, 0, Math.PI*2); ctx.fill();
      }
    });
  }

  // latest point
  const last = v[n-1];
  const bad = o.band && (last < o.band[0] || last > o.band[1]);
  ctx.fillStyle = bad ? COL.red : o.color;
  ctx.beginPath(); ctx.arc(X(n-1), Y(last), 4.5, 0, Math.PI*2); ctx.fill();
  ctx.strokeStyle = COL.bg; ctx.lineWidth = 2; ctx.stroke();

  // hover
  if (s.h !== null && s.h !== undefined && s.h < n) {
    const i = s.h, x = X(i), y = Y(v[i]);
    ctx.strokeStyle = COL.muted; ctx.lineWidth = 1;
    ctx.beginPath(); ctx.moveTo(x,T); ctx.lineTo(x,T+ch); ctx.stroke();
    ctx.fillStyle = COL.text;
    ctx.beginPath(); ctx.arc(x,y,4.5,0,Math.PI*2); ctx.fill();

    const ago = n - 1 - i;
    const t1 = ago === 0 ? "Latest" : (ago*IV >= 60 ? (Math.round(ago*IV/6)/10) + " min ago" : (ago*IV) + " s ago");
    const t2 = v[i] + " " + (o.unit || "");

    ctx.font = "600 12px " + FONT;
    const tw = Math.max(ctx.measureText(t1).width, ctx.measureText(t2).width) + 20;
    let bx = x + 12; if (bx + tw > w - 4) bx = x - 12 - tw;
    const by = Math.max(T, Math.min(y - 20, T + ch - 44));

    ctx.fillStyle = COL.bg; ctx.strokeStyle = COL.line; ctx.lineWidth = 1;
    ctx.beginPath();
    if (ctx.roundRect) ctx.roundRect(bx,by,tw,40,8); else ctx.rect(bx,by,tw,40);
    ctx.fill(); ctx.stroke();

    ctx.fillStyle = COL.muted; ctx.textAlign = "left"; ctx.font = "11px " + FONT;
    ctx.fillText(t1, bx + 10, by + 15);
    ctx.fillStyle = COL.text; ctx.font = "600 13px " + FONT;
    ctx.fillText(t2, bx + 10, by + 31);
  }
}

function setupChart(id, opts) {
  charts[id] = { v:[], o:opts, h:null, geo:{ L:34, cw:100 } };
  const cv = $(id);

  function move(clientX) {
    const s = charts[id], n = s.v.length;
    if (n === 0) return;
    const r = cv.getBoundingClientRect();
    const x = clientX - r.left;
    const i = n < 2 ? 0 : Math.round((x - s.geo.L)/s.geo.cw*(n-1));
    s.h = Math.max(0, Math.min(n-1, i));
    drawChart(id);
  }
  function leave() { charts[id].h = null; drawChart(id); }

  cv.addEventListener("mousemove", e => move(e.clientX));
  cv.addEventListener("mouseleave", leave);
  cv.addEventListener("touchstart", e => move(e.touches[0].clientX), { passive:true });
  cv.addEventListener("touchmove",  e => move(e.touches[0].clientX), { passive:true });
  cv.addEventListener("touchend", leave);
}

function showStats(id, vals, unit, dec) {
  const el = $(id);
  if (!vals.length) { el.innerHTML = ""; return; }
  const mn = Math.min.apply(null, vals), mx = Math.max.apply(null, vals);
  const av = vals.reduce((a,b) => a + b, 0)/vals.length;
  const box = (l,x) => '<div class="stat"><small>' + l + '</small><b>' + x.toFixed(dec) + unit + '</b></div>';
  el.innerHTML = box("Min",mn) + box("Avg",av) + box("Max",mx) + box("Now",vals[vals.length-1]);
}

        if (!r.ok) throw 0;
        toast("Saved");
        updateDashboard();
      } catch (e) { toast("Could not reach the ESP32"); }
    });
  });
}
function loadSliders(c) {
  SLIDERS.forEach(p => {
    const s = $(p[0]);
    if (document.activeElement === s) return;
    s.value = c[p[0]]; $(p[1]).innerText = c[p[0]];
  });
}
async function calibrate(type) {
  try {
    const r = await fetch("/cal?type=" + type);
    const j = await r.json();
    if (j.ok) toast("Soil " + type + " point saved (sensor " + j.raw + ")");
    else toast("Not saved (sensor " + j.raw + "). The dry value must be higher than the wet value.");
    updateDashboard();
  } catch (e) { toast("Could not reach the ESP32"); }
}

/* ---------- LED / BUZZER MODES ---------- */
const MODE_NAMES = ["Auto", "On", "Off"];

function initModes() {
  [["ledSeg","led","LED"], ["buzSeg","buz","Buzzer"]].forEach(p => {
    $(p[0]).querySelectorAll("button").forEach(b => {
      b.addEventListener("click", async () => {
        try {
          const r = await fetch("/mode?" + p[1] + "=" + b.dataset.v);
          if (!r.ok) throw 0;
          toast(p[2] + " set to " + MODE_NAMES[+b.dataset.v]);
          updateDashboard();
        } catch (e) { toast("Could not reach the ESP32"); }
      });
    });
  });
}
function showModes(c) {
  [["ledSeg", c.ledMode], ["buzSeg", c.buzMode]].forEach(p => {
    $(p[0]).querySelectorAll("button").forEach(b => {
      const on = (+b.dataset.v === p[1]);
      b.className = on ? ("active" + (p[1] === 2 ? " off" : "")) : "";
    });
  });
}

/* ---------- MAIN UPDATE ---------- */
async function updateDashboard() {
  try {
    const ctl = new AbortController();
    const tm = setTimeout(() => ctl.abort(), 8000);
    const response = await fetch("/data", { signal:ctl.signal, cache:"no-store" });
    clearTimeout(tm);
    if (!response.ok) throw new Error("Server error");
    const d = await response.json();

    cfg = d.cfg;
    IV = d.interval || 10;

    if ((d.refresh && d.refresh !== REFRESH) || !refreshTimer) {
      REFRESH = d.refresh || 20;
      clearInterval(refreshTimer);
      refreshTimer = setInterval(updateDashboard, REFRESH * 1000);
    }

    loadSliders(cfg);
    showModes(cfg);
    updateCfgLines();
    nextAt = Date.now() + d.next * 1000;

    // Temperature
    let tCls, tTxt;
    if (d.dhtError) { tTxt = "Check DHT11"; tCls = "bad"; }
    else if (d.temperature < cfg.tempLow) { tTxt = "Too cold"; tCls = "cold"; }
    else if (d.temperature > cfg.tempHigh) { tTxt = "Too hot"; tCls = "bad"; }
    else { tTxt = "Normal"; tCls = "ok"; }
    chip("tempStatus", tTxt, tCls);
    $("tempVal").innerText = d.dhtError ? "--" : d.temperature.toFixed(1);
    setMeter("temp", d.temperature, 0, 50, cfg.tempLow, cfg.tempHigh, tCls);
    $("tempNote").innerText = "Safe range " + cfg.tempLow + " to " + cfg.tempHigh + " \u00B0C";

    // Humidity
    let hCls, hTxt;
    if (d.dhtError) { hTxt = "Check DHT11"; hCls = "bad"; }
    else if (d.humidity < cfg.humLow) { hTxt = "Too dry"; hCls = "warn"; }
    else if (d.humidity > cfg.humHigh) { hTxt = "Too humid"; hCls = "bad"; }
    else { hTxt = "Normal"; hCls = "ok"; }
    chip("humStatus", hTxt, hCls);
    $("humVal").innerText = d.dhtError ? "--" : d.humidity.toFixed(0);
    setMeter("hum", d.humidity, 0, 100, cfg.humLow, cfg.humHigh, hCls);
    $("humNote").innerText = "Safe range " + cfg.humLow + " to " + cfg.humHigh + " %";

    // Soil
    let sCls, sTxt;
    if (d.soil < cfg.soilAlert) { sTxt = "Needs water"; sCls = "bad"; }
    else if (d.soil <= 70) { sTxt = "Good"; sCls = "ok"; }
    else { sTxt = "Wet"; sCls = "cold"; }
    chip("soilStatus", sTxt, sCls);
    $("soilVal").innerText = d.soil;
    setMeter("soil", d.soil, 0, 100, cfg.soilAlert, 100, sCls);
    $("soilNote").innerText = "Alert below " + cfg.soilAlert + " %, sensor reads " + d.soilRaw;

    // Rain (lower sensor value means wetter)
    const wet = (4095 - d.rainRaw)/4095*100;
    const trip = (4095 - cfg.rainThr)/4095*100;
    $("rainText").innerText = d.rain ? "Raining" : "No rain";
    chip("rainStatus", d.rain ? "Rain alert" : "Clear", d.rain ? "bad" : "ok");
    setMeter("rain", wet, 0, 100, 0, trip, d.rain ? "bad" : "ok");
    $("rainNote").innerText = "Sensor reads " + d.rainRaw + ", rain below " + cfg.rainThr;

    // Banner
    const b = $("banner");
    const reasons = [];
    if (d.rain) reasons.push("Rainfall detected");
    if (d.soil < cfg.soilAlert) reasons.push("Soil moisture below " + cfg.soilAlert + "%");

    const warns = [];
    if (d.dhtError) warns.push("DHT11 sensor error, check the wiring");
    else {
      if (d.temperature < cfg.tempLow) warns.push("Temperature below " + cfg.tempLow + "\u00B0C");
      if (d.temperature > cfg.tempHigh) warns.push("Temperature above " + cfg.tempHigh + "\u00B0C");
      if (d.humidity < cfg.humLow) warns.push("Humidity below " + cfg.humLow + "%");
      if (d.humidity > cfg.humHigh) warns.push("Humidity above " + cfg.humHigh + "%");
    }

    if (d.alert) {
      b.className = "alert";
      $("bIcon").innerText = "!"; $("bTitle").innerText = "Alert";
      $("bText").innerText = reasons.concat(warns).join(", ");
    } else if (warns.length) {
      b.className = "warn";
      $("bIcon").innerText = "!"; $("bTitle").innerText = "Warning";
      $("bText").innerText = warns.join(", ");
    } else {
      b.className = "";
      $("bIcon").innerText = "\u2713"; $("bTitle").innerText = "All conditions normal";
      $("bText").innerText = "Soil, rain, temperature and humidity are within limits";
    }

    $("ledOut").innerText = d.led ? "LED on" : "LED off";
    $("ledOut").className = "out" + (d.led ? " on" : "");
    $("buzOut").innerText = d.buzzer ? "Buzzer on" : "Buzzer off";
    $("buzOut").className = "out" + (d.buzzer ? " on" : "");

    // Charts
    const rec = d.history;
    const temps = rec.map(x => x.temp), hums = rec.map(x => x.hum), soils = rec.map(x => x.soil);
    updateChart("tempChart", temps);
    updateChart("humChart", hums);
    updateChart("soilChart", soils);
    showStats("tempStats", temps, "\u00B0C", 1);
    showStats("humStats", hums, "%", 1);
    showStats("soilStats", soils, "%", 0);

    renderRecent(rec);

    // System strip
    $("count").innerText = rec.length + " of 60";
    $("time").innerText = new Date().toLocaleTimeString();
    $("rssi").innerText = d.rssi + " dBm";
    $("uptime").innerText = fmtUp(d.uptime);
    $("ivText").innerText = "Every " + IV + " s";
    $("rfText").innerText = "Every " + REFRESH + " s";

    $("connection").innerText = "ESP32 connected";
    $("dot").className = "dot on";
  }
  catch (e) {
    $("connection").innerText = "ESP32 disconnected";
    $("dot").className = "dot off";
    $("bTitle").innerText = "Connection lost";
    $("bText").innerText = "Trying again every " + REFRESH + " seconds";
    $("bIcon").innerText = "!";
    $("banner").className = "warn";
  }
}

// ========== SETUP ==========
void setup() {

  Serial.begin(115200);

  pinMode(RAIN_PIN, INPUT);
  pinMode(SOIL_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);

  digitalWrite(BUZZER_PIN, LOW);
  digitalWrite(LED_PIN, LOW);

  analogReadResolution(12);

  loadSettings();
  dht.begin();

  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);            // faster, more reliable web responses
  WiFi.setAutoReconnect(true);
  WiFi.begin(ssid, password);

  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi Connected!");
  Serial.print("Website link: http://");
  Serial.println(WiFi.localIP());

  server.on("/", []() {
    server.send_P(200, "text/html", webpage);
  });
  server.on("/data", sendData);
  server.on("/set", handleSet);
  server.on("/mode", handleMode);
  server.on("/cal", handleCal);

  server.begin();
  Serial.println("Web Server Started!");

  readSensors();
  lastSample = millis();

  // Telegram startup message
  sendTelegram("Farm monitor online.\nDashboard: http://" + WiFi.localIP().toString());
}

// ========== MAIN LOOP ==========
void loop() {

  server.handleClient();

  unsigned long now = millis();

  // Reconnect WiFi if it drops
  if (WiFi.status() != WL_CONNECTED && now - lastWifiTry > 10000) {
    lastWifiTry = now;
    WiFi.reconnect();
  }

  if (now - lastSample >= SAMPLE_INTERVAL) {
    lastSample = now;
    readSensors();
  }

  updateAlerts();
}
